# Database spill backends are optional: each one is compiled in only when its
# client library is found. The realurl library itself never links them.
find_package(SQLite3)
find_package(PostgreSQL)
if(NOT PostgreSQL_FOUND)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(LIBPQ QUIET IMPORTED_TARGET libpq)
  endif()
endif()

add_library(realurl_bench_spill STATIC spill_backends.cpp)
# Only needs the SpillStore header, not the library (and its link graph).
target_include_directories(realurl_bench_spill PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../include")
target_link_libraries(realurl_bench_spill PUBLIC Threads::Threads PRIVATE realurl_warnings)
if(SQLite3_FOUND)
  if(TARGET SQLite3::SQLite3)
    target_link_libraries(realurl_bench_spill PRIVATE SQLite3::SQLite3)
  else()
    target_link_libraries(realurl_bench_spill PRIVATE SQLite::SQLite3)
  endif()
  target_compile_definitions(realurl_bench_spill PRIVATE REALURL_BENCH_SQLITE)
endif()
if(PostgreSQL_FOUND)
  target_link_libraries(realurl_bench_spill PRIVATE PostgreSQL::PostgreSQL)
  target_compile_definitions(realurl_bench_spill PRIVATE REALURL_BENCH_POSTGRES)
elseif(TARGET PkgConfig::LIBPQ)
  target_link_libraries(realurl_bench_spill PRIVATE PkgConfig::LIBPQ)
  target_compile_definitions(realurl_bench_spill PRIVATE REALURL_BENCH_POSTGRES)
endif()

add_executable(realurl_bench bench.cpp)
target_include_directories(realurl_bench PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../tests")
target_link_libraries(realurl_bench PRIVATE realurl::server realurl_test_support realurl_bench_spill realurl_warnings)
target_compile_definitions(realurl_bench PRIVATE REALURL_CERT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../tests/certs")

add_executable(spill_backends_test spill_backends_test.cpp)
target_link_libraries(spill_backends_test PRIVATE realurl_bench_spill realurl_warnings Threads::Threads)
